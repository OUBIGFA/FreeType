cmake_minimum_required(VERSION 3.24)
project(FreeTypeEnhancement LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_FMA ON CACHE BOOL "" FORCE)
set(GGML_F16C ON CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE ON CACHE BOOL "" FORCE)
add_subdirectory("${GGML_SOURCE}" ggml)
add_library(FreeTypeEnhancement SHARED gpt2.cpp)
target_link_libraries(FreeTypeEnhancement PRIVATE ggml ggml-cpu ggml-base)
target_compile_definitions(FreeTypeEnhancement PRIVATE NOMINMAX _USE_MATH_DEFINES)
target_compile_options(FreeTypeEnhancement PRIVATE /utf-8 /O2 /EHsc)
